    async function waitForFrames(el, timeoutMs = 8000) {
      const t0 = performance.now();
      el.muted = true;
      el.playsInline = true;
      if (el.readyState < 1) {
        await new Promise((resolve, reject) => {
          const t = setTimeout(() => reject(new Error("no frames")), timeoutMs);
          el.onloadedmetadata = () => {
            clearTimeout(t);
            resolve();
          };
        });
      }
      try { await el.play(); } catch (_) {}
      while (!el.videoWidth) {
        if (performance.now() - t0 > timeoutMs) throw new Error("no frames");
        await new Promise((r) => requestAnimationFrame(r));
      }
    }

    function detectMove(lm) {
      const nose = lm[0], ls = lm[11], rs = lm[12], lw = lm[15], rw = lm[16];
      if (!nose || !ls || !rs) return "still";
      const shoulderY = (ls.y + rs.y) / 2;
      const vis = (p) => !p || p.visibility === undefined || p.visibility > 0.5;
      const wristUp =
        (lw && vis(lw) && lw.y < shoulderY - 0.12) ||
        (rw && vis(rw) && rw.y < shoulderY - 0.12);
      waveFrames = wristUp ? waveFrames + 1 : 0;
      noseHist.push(nose.y);
      if (noseHist.length > 18) noseHist.shift();
      const span = noseHist.length >= 12 ? Math.max(...noseHist) - Math.min(...noseHist) : 0;
      nodFrames = span > 0.045 ? nodFrames + 1 : Math.max(0, nodFrames - 1);
      if (waveFrames >= 6) return "wave";
      if (nodFrames >= 8) return "nod";
      return "still";
    }
