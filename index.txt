    <button id="imagineBtn" class="ghost" disabled>Imagine me again</button>
    <span class="stat">Score <b id="score">0</b> · You <b id="youMove">—</b> · Clone <b id="cloneMove">—</b> · <span id="statusLine">cam off</span></span>

    <canvas id="overlay"></canvas>
    <div class="stamp" id="stamp">IT COPIED YOU</div>

  <p class="err" id="err"></p>

  <script type="module">
    import { PoseLandmarker, FilesetResolver } from "https://cdn.jsdelivr.net/npm/@mediapipe/tasks-vision@0.10.14";

    const video = document.getElementById("cam");
    const canvas = document.getElementById("overlay");
    const ctx = canvas.getContext("2d");
    const stamp = document.getElementById("stamp");
    const errEl = document.getElementById("err");
    const statusEl = document.getElementById("statusLine");
    const startBtn = document.getElementById("startBtn");
    const imagineBtn = document.getElementById("imagineBtn");

    const looks = ["poster", "ink", "warm", "cool"];
    let look = 0;

    let pose = null;
    let running = false;
    let lastLm = null;
    let score = 0;
    let lastScoreAt = 0;
    let userMove = "still";
    let cloneMove = "still";
    let cloneLockUntil = 0;
    let noseHist = [];
    let waveFrames = 0;
    let nodFrames = 0;

    async function waitForFrames(el, timeoutMs = 8000) {
      const t0 = performance.now();
      el.muted = true;
      el.playsInline = true;
      if (el.readyState < 1) {
        await new Promise((resolve, reject) => {
          const t = setTimeout(() => reject(new Error("no frames")), timeoutMs);
          el.onloadedmetadata = () => {
            clearTimeout(t);
            resolve();
          };
        });
      }
      try { await el.play(); } catch (_) {}
      while (!el.videoWidth) {
        if (performance.now() - t0 > timeoutMs) throw new Error("no frames");
        await new Promise((r) => requestAnimationFrame(r));
      }
    }

    function cartoonize(srcCanvas) {
      const s = srcCanvas.getContext("2d");
      const img = s.getImageData(0, 0, srcCanvas.width, srcCanvas.height);
      const d = img.data;
      const mode = looks[look];
      for (let i = 0; i < d.length; i += 4) {
        let r = d[i], g = d[i + 1], b = d[i + 2];
        const l = 0.32 * r + 0.5 * g + 0.18 * b;
        if (mode === "ink") {
          const v = l > 110 ? 235 : 30;
          r = g = b = v;
        } else if (mode === "warm") {
          r = Math.min(255, r * 1.25 + 20);
          b *= 0.75;
        } else if (mode === "cool") {
          b = Math.min(255, b * 1.25 + 16);
          r *= 0.8;
        } else {
          const crush = (v) => Math.round(Math.min(255, v * 1.15 + (v - l) * 0.55) / 42) * 42;
          r = crush(r); g = crush(g); b = crush(b);
        }
        d[i] = r; d[i + 1] = g; d[i + 2] = b;
      }
      s.putImageData(img, 0, 0);
      return srcCanvas;
    }

    function detectMove(lm) {
      const nose = lm[0], ls = lm[11], rs = lm[12], lw = lm[15], rw = lm[16];
      if (!nose || !ls || !rs) return "still";

      const shoulderY = (ls.y + rs.y) / 2;
      const vis = (p) => !p || p.visibility === undefined || p.visibility > 0.5;
      const wristUp =
        (lw && vis(lw) && lw.y < shoulderY - 0.12) ||
        (rw && vis(rw) && rw.y < shoulderY - 0.12);

      waveFrames = wristUp ? waveFrames + 1 : 0;
      noseHist.push(nose.y);
      if (noseHist.length > 18) noseHist.shift();
      const span = noseHist.length >= 12 ? Math.max(...noseHist) - Math.min(...noseHist) : 0;
      nodFrames = span > 0.045 ? nodFrames + 1 : Math.max(0, nodFrames - 1);

      if (waveFrames >= 6) return "wave";
      if (nodFrames >= 8) return "nod";
      return "still";
    }

    function paint() {
      const w = canvas.clientWidth || 640;
      const h = canvas.clientHeight || 480;
      canvas.width = w;
      canvas.height = h;

      ctx.save();
      ctx.translate(w, 0);
      ctx.scale(-1, 1);
      if (video.videoWidth) ctx.drawImage(video, 0, 0, w, h);
      else {
        ctx.fillStyle = "#222";
        ctx.fillRect(0, 0, w, h);
      }
      ctx.restore();

      const vw = video.videoWidth;
      const vh = video.videoHeight;
      if (!vw || !vh) return;

      const now = performance.now();
      const cloneW = Math.round(Math.min(w, h) * 0.42);
      const cloneH = Math.round(cloneW * 1.25);

      let x = w * 0.78;
      let y = h * 0.38;
      if (lastLm && lastLm[12]) {
        x = Math.min(w - cloneW * 0.55, Math.max(cloneW * 0.45, (1 - lastLm[12].x) * w + cloneW * 0.35));
        y = Math.min(h * 0.48, Math.max(h * 0.28, lastLm[12].y * h + cloneH * 0.05));
      }

      const bounce = cloneMove === "wave" ? Math.sin(now / 140) * 10 : 0;
      const tilt = cloneMove === "nod" ? Math.sin(now / 160) * 0.12 : Math.sin(now / 400) * 0.03;

      const crop = document.createElement("canvas");
      crop.width = 200;
      crop.height = 250;
      const c = crop.getContext("2d");
      const sideW = Math.min(vw, vh) * 0.55;
      const sideH = sideW * 1.25;
      const sx = Math.max(0, (vw - sideW) / 2);
      const sy = Math.max(0, vh * 0.04);
      c.translate(crop.width, 0);
      c.scale(-1, 1);
      c.drawImage(video, sx, sy, sideW, sideH, 0, 0, crop.width, crop.height);
      cartoonize(crop);

      ctx.save();
      ctx.translate(x, y + bounce);
      ctx.rotate(tilt);
      ctx.beginPath();
      if (ctx.roundRect) ctx.roundRect(-cloneW / 2, -cloneH / 2, cloneW, cloneH, 28);
      else ctx.rect(-cloneW / 2, -cloneH / 2, cloneW, cloneH);
      ctx.clip();
      ctx.drawImage(crop, -cloneW / 2, -cloneH / 2, cloneW, cloneH);
      ctx.restore();

      ctx.save();
      ctx.translate(x, y + bounce);
      ctx.rotate(tilt);
      ctx.strokeStyle = "rgba(255,255,255,.4)";
      ctx.lineWidth = 3;
      ctx.beginPath();
      if (ctx.roundRect) ctx.roundRect(-cloneW / 2, -cloneH / 2, cloneW, cloneH, 28);
      else ctx.rect(-cloneW / 2, -cloneH / 2, cloneW, cloneH);
      ctx.stroke();
      ctx.restore();
    }

    function tick() {
      if (!running) return;
      const now = performance.now();
      statusEl.textContent = video.videoWidth ? video.videoWidth + "x" + video.videoHeight : "no frames";

      if (pose && video.videoWidth) {
        try {
          const res = pose.detectForVideo(video, now);
          const lm = res.landmarks && res.landmarks[0];
          if (lm) lastLm = lm;
          userMove = lm ? detectMove(lm) : "still";
        } catch (_) {}
      }

      if (now > cloneLockUntil) {
        cloneMove = Math.random() < 0.5 ? "wave" : "nod";
        cloneLockUntil = now + 900 + Math.random() * 700;
      }

      if (
        cloneMove !== "still" &&
        userMove === cloneMove &&
        now < cloneLockUntil &&
        now - lastScoreAt > 1200
      ) {
        score += 1;
        lastScoreAt = now;
        document.getElementById("score").textContent = score;
        stamp.classList.add("on");
        setTimeout(() => stamp.classList.remove("on"), 700);
      }

      document.getElementById("youMove").textContent = userMove;
      document.getElementById("cloneMove").textContent = cloneMove;
      paint();
      requestAnimationFrame(tick);
    }

    startBtn.addEventListener("click", async () => {
      startBtn.disabled = true;
      errEl.textContent = "Starting camera…";
      try {
        const stream = await navigator.mediaDevices.getUserMedia({
          video: { width: { ideal: 1280 }, height: { ideal: 720 }, facingMode: "user" },
          audio: false
        });
        video.srcObject = stream;
        await waitForFrames(video);

        running = true;
        errEl.textContent = "";
        tick();

        try {
          errEl.textContent = "Loading pose model…";
          const files = await FilesetResolver.forVisionTasks(
            "https://cdn.jsdelivr.net/npm/@mediapipe/tasks-vision@0.10.14/wasm"
          );
          pose = await PoseLandmarker.createFromOptions(files, {
            baseOptions: {
              modelAssetPath:
                "https://storage.googleapis.com/mediapipe-models/pose_landmarker/pose_landmarker_lite/float16/1/pose_landmarker_lite.task"
            },
            runningMode: "VIDEO",
            numPoses: 1
          });
          imagineBtn.disabled = false;
          errEl.textContent = "Look: " + looks[look];
        } catch (modelErr) {
          console.error(modelErr);
          errEl.textContent = "Camera is on. Pose model failed: " + (modelErr && modelErr.message ? modelErr.message : modelErr);
        }
      } catch (e) {
        console.error(e);
        errEl.textContent = "Camera blocked or sent no picture. Allow camera for this site, close Zoom/Meet, reload.";
        startBtn.disabled = false;
        running = false;
        if (video.srcObject) {
          video.srcObject.getTracks().forEach((t) => t.stop());
          video.srcObject = null;
        }
      }
    });

    imagineBtn.addEventListener("click", () => {
      look = (look + 1) % looks.length;
      errEl.textContent = "Look: " + looks[look];
    });
  </script>
